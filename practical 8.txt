Program Statement:

Beginning with an empty binary search tree, Construct a binary search tree by inserting
the values in the order given. After constructing a binary tree -
i.	Insert new node
ii.	Find number of nodes in longest path from root
iii.	Minimum data value found in the tree
iv.	Change a tree so that the roles of the left and right pointers are swapped at every node
v.	Search a value.

#include <stdio.h>
#include <stdlib.h>

// Define the structure of a node
struct Node {
    int data;
    struct Node* left;
    struct Node* right;
};

// Create a new node
struct Node* createNode(int data) {
    struct Node* newNode = (struct Node*)malloc(sizeof(struct Node));
    newNode->data = data;
    newNode->left = newNode->right = NULL;
    return newNode;
}

// Insert a node into BST
struct Node* insert(struct Node* root, int data) {
    if (root == NULL)
        return createNode(data);
    if (data < root->data)
        root->left = insert(root->left, data);
    else if (data > root->data)
        root->right = insert(root->right, data);
    return root;
}

// Find height of the tree (Longest path)
int longestPath(struct Node* root) {
    if (root == NULL)
        return 0;
    int left = longestPath(root->left);
    int right = longestPath(root->right);
    return (left > right ? left : right) + 1;
}

// Find minimum value in BST
int findMin(struct Node* root) {
    if (root == NULL) {
        printf("Tree is empty.\n");
        return -1;
    }
    while (root->left != NULL)
        root = root->left;
    return root->data;
}

// Mirror the tree
void mirrorTree(struct Node* root) {
    if (root == NULL) return;
    mirrorTree(root->left);
    mirrorTree(root->right);
    
    // Swap left and right
    struct Node* temp = root->left;
    root->left = root->right;
    root->right = temp;
}

// Search a value
int search(struct Node* root, int key) {
    if (root == NULL)
        return 0;
    if (key == root->data)
        return 1;
    else if (key < root->data)
        return search(root->left, key);
    else
        return search(root->right, key);
}

// In-order traversal
void inorder(struct Node* root) {
    if (root == NULL)
        return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

// Driver code
int main() {
    struct Node* root = NULL;
    int initial[] = {50, 30, 20, 40, 70, 60, 80};
    int n = sizeof(initial)/sizeof(initial[0]);

    // Construct initial tree
    for (int i = 0; i < n; i++)
        root = insert(root, initial[i]);

    printf("In-order traversal of BST: ");
    inorder(root);
    printf("\n");

    // Insert new node
    printf("Inserting 25...\n");
    root = insert(root, 25);

    printf("In-order traversal after insertion: ");
    inorder(root);
    printf("\n");

    // Longest path (height)
    printf("Longest path (height) from root: %d\n", longestPath(root));

    // Minimum value
    printf("Minimum value in BST: %d\n", findMin(root));

    // Search for a value
    int key = 60;
    printf("Searching for %d: %s\n", key, search(root, key) ? "Found" : "Not Found");

    // Mirror the tree
    mirrorTree(root);
    printf("In-order traversal after mirroring: ");
    inorder(root);
    printf("\n");

    return 0;
}

Minimum value in BST: 20
Searching for 60: Found
