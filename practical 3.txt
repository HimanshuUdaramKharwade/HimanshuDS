Consider the telephone book database of N clients. Make use of a hash table implementation to quickly look up a client’s telephone number. Make use of linear probing, double hashing and quadratic collision handling techniques.

#include <iostream>
#include <vector>
#include <string>
using namespace std;

class TelephoneBook {
    vector<pair<string,string>> table;
    int size;

    // Primary hash: polynomial rolling mod table size
    int hash_function(const string& key) {
        long long h = 0;
        for (char c : key) h = (h * 31 + (unsigned char)c) % size;
        return (int)h;
    }

    // Second hash for double hashing (must be non-zero and < size)
    int second_hash(const string& key) {
        int prime = (size > 3) ? size - 2 : 2; // a small prime < size (simple choice)
        int h = 0;
        for (char c : key) h = (h * 17 + (unsigned char)c) % prime;
        int val = h % prime;
        return (val == 0) ? 1 : val;
    }

public:
    TelephoneBook(int s) : size(s) { table.resize(s, {"", ""}); }

    // Insert or update using chosen method:
    // method = 1 -> linear, 2 -> quadratic, 3 -> double hashing
    void insert(const string& name, const string& phone, int method = 1) {
        int h = hash_function(name);
        int h2 = second_hash(name);
        for (int i = 0; i < size; ++i) {
            int idx;
            if (method == 1) { // linear
                idx = (h + i) % size;
            } else if (method == 2) { // quadratic
                idx = (h + i + i * i) % size;
            } else { // double hashing
                idx = (h + i * h2) % size;
            }

            // empty slot or same name -> place/update
            if (table[idx].first == "" || table[idx].first == name) {
                table[idx] = {name, phone};
                return;
            }
        }
        cout << "Insert failed: table full\n";
    }

    // Search using same probing method as insertion
    string search(const string& name, int method = 1) {
        int h = hash_function(name);
        int h2 = second_hash(name);
        for (int i = 0; i < size; ++i) {
            int idx;
            if (method == 1) idx = (h + i) % size;
            else if (method == 2) idx = (h + i + i * i) % size;
            else idx = (h + i * h2) % size;

            if (table[idx].first == "") return "Not Found";      // empty -> not present
            if (table[idx].first == name) return table[idx].second;
        }
        return "Not Found";
    }

    void display() {
        cout << "\nIndex : Name -> Phone\n";
        for (int i = 0; i < size; ++i) {
            if (table[i].first != "")
                cout << i << " : " << table[i].first << " -> " << table[i].second << "\n";
            else
                cout << i << " : Empty\n";
        }
    }
};

int main() {

    TelephoneBook phonebook(size);

    while (true) {
